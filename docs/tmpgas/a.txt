<script>
let S={},ALL=[],CUR='';
const $=id=>document.getElementById(id);
const esc=s=>String(s==null?'':s).replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
function toast(t){const m=$('msg');m.textContent=t;m.style.display='block';setTimeout(()=>m.style.display='none',1800)}
function run(fn,...a){return new Promise((ok,no)=>google.script.run.withSuccessHandler(r=>ok(typeof r==='string'&&r[0]==='{'?JSON.parse(r):r)).withFailureHandler(e=>{toast('⚠️ '+e.message);no(e)})[fn](...a))}
function drawState(){
  const on=S.status==='open';
  $('st').innerHTML=on?`<b>공개 중</b> · 시작 ${esc(S.start)} · 오늘 <b>${S.today}일차</b> (한 바퀴 ${S.cycle}일 중 ${S.cday}일차)`:`<b style="color:#999">공개 꺼짐</b>${S.start?' · 시작일 '+esc(S.start)+' (이어서 시작 가능)':''}`;
  $('links').innerHTML=`<a href="${esc(S.url)}" target="_blank">📄 시트 열기</a> · 문항 ${S.total}개 중 공개 ${S.open}개`;
  const sel=$('dsel'),keep=sel.value;sel.innerHTML='';
  for(let d=1;d<=S.cycle;d++){const o=document.createElement('option');o.value=d;o.textContent=d+'일차'+(d===S.cday&&on?' (오늘)':'');sel.appendChild(o)}
  sel.value=keep||S.cday||1;
}
async function act(a,n){S=await run('adminAction',a,n);drawState();toast('적용했어요');loadDay()}
async function loadStats(){
  const r=await run('adminStats');
  if(!r.days.length){$('stats').textContent='아직 기록이 없어요.';return}
  const mx=Math.max(...r.days.map(d=>d.view),1);
  $('stats').innerHTML=r.days.map(d=>`<div class="li">${esc(d.date)} · 방문자 <b>${d.visitors}</b> · 조회 ${d.view} · 답변확인 ${d.check}<br><span class="bar" style="width:${Math.round(d.view/mx*100)}%"></span></div>`).join('')
   +(r.top.length?`<div style="margin-top:8px"><b>답변 확인 많은 문제</b><br>${r.top.map(t=>esc(t.id)+' ('+t.n+')').join(' · ')}</div>`:'');
}
async function loadDay(){
  const n=+$('dsel').value||1;const r=await run('adminDay',n);ALL=r.all;
  $('day').innerHTML=r.items.length?r.items.map(x=>`<div class="li"><span class="tag ${x.kind==='계산'?'c':''}">${esc(x.kind)}</span><b>${esc(x.id)}</b> ${esc(x.q)} <button class="g" style="padding:2px 8px" onclick="swap('${esc(x.id)}','')">빼기</button> <button class="g" style="padding:2px 8px" onclick="edit('${esc(x.id)}')">고치기</button></div>`).join(''):'<div class="muted">이 일차에는 문제가 없어요.</div>';
  $('addsel').innerHTML=ALL.filter(x=>x.open).map(x=>`<option value="${esc(x.id)}">${esc(x.id)} [${esc(x.kind)}] ${esc(x.q)} (현재 ${esc(x.days)||'없음'})</option>`).join('');
  drawAll();
}
async function swap(out,inn){await run('adminSwap',+$('dsel').value,out,inn);toast('바꿨어요');loadDay()}
function addQ(){swap('',$('addsel').value)}
function drawAll(){
  const f=$('flt').value.trim();
  $('all').innerHTML=ALL.filter(x=>!f||x.id.includes(f)||x.q.includes(f)).map(x=>`<div class="li"><span class="tag ${x.kind==='계산'?'c':''} ${x.open?'':'off'}">${x.open?esc(x.kind):'숨김'}</span><b>${esc(x.id)}</b> ${esc(x.q)}… <span class="muted">${esc(x.days)}일차</span> <button class="g" style="padding:2px 8px" onclick="edit('${esc(x.id)}')">미리보기·고치기</button></div>`).join('');
}
async function edit(id){
  const r=await run('adminItem',id);CUR=id;$('edt').textContent='✏️ '+id+' 고치기';
  ['q','s','a','k','kt','tip'].forEach(k=>$('e_'+k).value=r[k]||'');$('e_days').value=(r.days||[]).join(',');$('e_open').checked=!!r.open;
  $('edbox').style.display='block';$('edbox').scrollIntoView({behavior:'smooth'});
}
async function saveQ(){
  const o={};['q','s','a','k','kt','tip','days'].forEach(k=>o[k]=$('e_'+k).value);o.open=$('e_open').checked;
  await run('adminSaveItem',CUR,o);toast('저장했어요');loadDay();
}
(async()=>{S=await run('adminState');drawState();loadStats();loadDay();})();
</script>
